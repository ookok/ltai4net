- **名称**：Minimal Todo
- **功能**：
  - 添加待办事项（标题、截止日期、优先级）

### 3.2 数据库模块（db.js）
使用 `better-sqlite3` 同步操作，避免回调复杂性，性能佳。
```javascript
const Database = require('better-sqlite3');
const path = require('path');

// 数据库文件存储在项目根目录
const db = new Database(path.join(__dirname, 'todo.db'));

// 启用 WAL 模式提高并发性能
db.pragma('journal_mode = WAL');

// 初始化表结构
db.exec(`
  CREATE TABLE IF NOT EXISTS todos (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    title TEXT NOT NULL,
    completed INTEGER DEFAULT 0,
    priority TEXT DEFAULT 'medium',
    due_date TEXT,
    created_at TEXT DEFAULT (datetime('now')),
    updated_at TEXT DEFAULT (datetime('now'))
  )
`);

// 提供的数据操作接口
const todoModel = {
  // 获取所有任务，支持筛选
  getAll: (filter = 'all') => {
    let query = 'SELECT * FROM todos';
    const params = [];
    if (filter === 'active') {
      query += ' WHERE completed = 0';
    } else if (filter === 'completed') {
      query += ' WHERE completed = 1';
    }
    query += ' ORDER BY created_at DESC';
    return db.prepare(query).all(...params);
  },

  // 创建新任务
  create: ({ title, priority = 'medium', due_date = null }) => {
    const stmt = db.prepare(`
      INSERT INTO todos (title, priority, due_date) 
      VALUES (?, ?, ?)
    `);
    const info = stmt.run(title, priority, due_date);
    return { id: info.lastInsertRowid, title, priority, due_date, completed: 0 };
  },

  // 切换完成状态
  toggle: (id) => {
    const todo = db.prepare('SELECT completed FROM todos WHERE id = ?').get(id);
    if (!todo) return null;
    const newStatus = todo.completed ? 0 : 1;
    db.prepare('UPDATE todos SET completed = ?, updated_at = datetime("now") WHERE id = ?')
      .run(newStatus, id);
    return { id, completed: !!newStatus };
  },

  // 删除任务
  delete: (id) => {
    const stmt = db.prepare('DELETE FROM todos WHERE id = ?');
    const info = stmt.run(id);
    return info.changes > 0;
  }
};

module.exports = todoModel;
```

### 3.3 Express 服务器（server.js）
提供 RESTful API 并托管前端静态文件。
```javascript
const express = require('express');
const path = require('path');
const todoModel = require('./db');

const app = express();
const PORT = 3000;

// 解析 JSON 请求体
app.use(express.json());

// 托管前端静态文件
app.use(express.static(path.join(__dirname, 'public')));

// API 路由
// 获取任务列表 ?filter=all|active|completed
app.get('/api/todos', (req, res) => {
  const filter = req.query.filter || 'all';
  const todos = todoModel.getAll(filter);
  res.json(todos);
});

// 创建任务
app.post('/api/todos', (req, res) => {
  const { title, priority, due_date } = req.body;
  if (!title || title.trim() === '') {
    return res.status(400).json({ error: 'Title is required' });
  }
  const newTodo = todoModel.create({ title: title.trim(), priority, due_date });
  res.status(201).json(newTodo);
});

// 切换任务完成状态
app.patch('/api/todos/:id/toggle', (req, res) => {
  const id = parseInt(req.params.id, 10);
  const result = todoModel.toggle(id);
  if (!result) {
    return res.status(404).json({ error: 'Todo not found' });
  }
  res.json(result);
});

// 删除任务
app.delete('/api/todos/:id', (req, res) => {
  const id = parseInt(req.params.id, 10);
  const success = todoModel.delete(id);
  if (!success) {
    return res.status(404).json({ error: 'Todo not found' });
  }
  res.status(204).send();
});

// 启动服务
app.listen(PORT, () => {
  console.log(`Server running at http://localhost:${PORT}`);
});
```

### 3.4 前端界面（public/index.html）
简洁的响应式单页应用，通过 fetch 与后端 API 交互。
```html
<!DOCTYPE html>
<html lang="zh-CN">
<head>
  <meta charset="UTF-8">
  <meta name="viewport" content="width=device-width, initial-scale=1.0">
  <title>待办事项 · Minimal Todo</title>
  <link rel="stylesheet" href="style.css">
</head>
<body>
  <div class="container">
    <h1>📝 待办事项</h1>

    <!-- 添加任务表单 -->
    <form id="todo-form">
      <input type="text" id="title-input" placeholder="添加新任务..." required />
      <select id="priority-input">
        <option value="low">低优先级</option>
        <option value="medium" selected>中优先级</option>
        <option value="high">高优先级</option>
      </select>
      <input type="date" id="due-date-input" />
      <button type="submit">添加</button>
    </form>

    <!-- 筛选按钮 -->
    <div class="filters">
      <button class="filter-btn active" data-filter="all">全部</button>
      <button class="filter-btn" data-filter="active">待完成</button>
      <button class="filter-btn" data-filter="completed">已完成</button>
    </div>

    <!-- 任务列表 -->
    <ul id="todo-list"></ul>
  </div>

  <script>
    const API_BASE = '/api/todos';
    let currentFilter = 'all';

    // 获取并渲染任务列表
    async function fetchTodos() {
      try {
        const res = await fetch(`${API_BASE}?filter=${currentFilter}`);
        const todos = await res.json();
        renderTodos(todos);
      } catch (error) {
        console.error('获取任务失败', error);
      }
    }

    // 渲染列表
    function renderTodos(todos) {
      const list = document.getElementById('todo-list');
      if (todos.length === 0) {
        list.innerHTML = '<li class="empty">暂无任务，快去添加吧~</li>';
        return;
      }
      list.innerHTML = todos.map(todo => `
        <li class="todo-item ${todo.completed ? 'completed' : ''}" data-id="${todo.id}">
          <span class="todo-check">${todo.completed ? '✅' : '⬜'}</span>
          <span class="todo-title">${escapeHtml(todo.title)}</span>
          <span class="todo-priority ${todo.priority}">${priorityLabel(todo.priority)}</span>
          ${todo.due_date ? `<span class="todo-date">📅 ${todo.due_date}</span>` : ''}
          <button class="toggle-btn">${todo.completed ? '↩️ 取消' : '✔️ 完成'}</button>
          <button class="delete-btn">🗑️ 删除</button>
        </li>
      `).join('');
    }

    function escapeHtml(text) {
      return text.replace(/[&<>"']/g, m => ({ '&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;',"'":'&#39;' })[m]);
    }

    function priorityLabel(p) {
      const map = { low: '🟢低', medium: '🟡中', high: '🔴高' };
      return map[p] || p;
    }

    // 事件监听（事件委托）
    document.getElementById('todo-list').addEventListener('click', async (e) => {
      const li = e.target.closest('.todo-item');
      if (!li) return;
      const id = li.dataset.id;
      if (e.target.classList.contains('toggle-btn')) {
        await fetch(`${API_BASE}/${id}/toggle`, { method: 'PATCH' });
        fetchTodos();
      } else if (e.target.classList.contains('delete-btn')) {
        if (!confirm('确定删除该任务？')) return;
        await fetch(`${API_BASE}/${id}`, { method: 'DELETE' });
        fetchTodos();
      }
    });

    // 添加任务表单提交
    document.getElementById('todo-form').addEventListener('submit', async (e) => {
      e.preventDefault();
      const titleInput = document.getElementById('title-input');
      const priorityInput = document.getElementById('priority-input');
      const dueDateInput = document.getElementById('due-date-input');
      const title = titleInput.value.trim();
      if (!title) return;

      await fetch(API_BASE, {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({
          title,
          priority: priorityInput.value,
          due_date: dueDateInput.value || null
        })
      });
      titleInput.value = '';
      dueDateInput.value = '';
      fetchTodos();
    });

    // 筛选按钮切换
    document.querySelectorAll('.filter-btn').forEach(btn => {
      btn.addEventListener('click', () => {
        document.querySelectorAll('.filter-btn').forEach(b => b.classList.remove('active'));
        btn.classList.add('active');
        currentFilter = btn.dataset.filter;
        fetchTodos();
      });
    });

    // 初始加载
    fetchTodos();
  </script>
</body>
</html>
```
